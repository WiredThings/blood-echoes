243: void TextureCache::InvalidateMemory(VAddr addr, size_t size) {
244:     std::scoped_lock lock{mutex};
245:     const auto pages_start = PageManager::GetPageAddr(addr);
246:     const auto pages_end = PageManager::GetNextPageAddr(addr + size - 1);
247:     ForEachImageInRegion(pages_start, pages_end - pages_start, [&](ImageId image_id, Image& image) {
248:         const auto image_begin = image.info.guest_address;
249:         const auto image_end = image.info.guest_address + image.info.guest_size;
250:         if (image.Overlaps(addr, size)) {
251:             // Modified region overlaps image, so the image was definitely accessed by this fault.
252:             // Untrack the image, so that the range is unprotected and the guest can write freely.
253:             image.flags |= ImageFlagBits::CpuDirty;
254:             PageManager::NoteImageFault(image.info.guest_address, image.info.guest_size,
255:                                         image.info.size.width, image.info.size.height,
256:                                         u32(image.info.pixel_format), u32(image.info.tile_mode));
257:             UntrackImage(image_id);
258:         } else if (pages_end < image_end) {
259:             // This page access may or may not modify the image.
260:             // We should not mark it as dirty now. If it really was modified
261:             // it will receive more invalidations on its other pages.
262:             // Remove tracking from this page only.
263:             UntrackImageHead(image_id);
264:         } else if (image_begin < pages_start) {
265:             // This page access does not modify the image but the page should be untracked.
266:             // We should not mark this image as dirty now. If it really was modified
267:             // it will receive more invalidations on its other pages.
268:             UntrackImageTail(image_id);
269:         } else {
270:             // Image begins and ends on this page so it can not receive any more invalidations.
271:             // We will check it's hash later to see if it really was modified.
272:             MarkAsMaybeDirty(image_id, image);
273:         }
274:     });
